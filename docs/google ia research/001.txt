Estructura del Core en JUCE para el Control BidireccionalPara gestionar esto de forma limpia en tu arquitectura por capas, necesitas dos componentes principales en C++: un receptor (juce::MidiInputCallback) y un emisor.Aquí tienes el esqueleto técnico de cómo estructurar la clase encargada de la bidireccionalidad:cpp#include <JuceHeader.h>

class AiraMidiController : public juce::MidiInputCallback,
                           public juce::MidiBuffer
{
public:
    AiraMidiController() 
    {
        // Inicializar dispositivos MIDI
        startTime = juce::Time::getMillisecondCounterHiRes() * 0.001;
    }

    // --- HARDWARE -> WEBUI (Recepción) ---
    void handleIncomingMidiMessage(juce::MidiInput* source, const juce::MidiMessage& message) override
    {
        if (message.isController())
        {
            const int ccNumber = message.getControllerNumber();
            const int ccValue = message.getControllerValue();
            const int channel = message.getChannel();

            // Evitamos procesar en el hilo de alta prioridad de MIDI.
            // Enviamos los datos de forma segura al MessageManager o tu hilo de WebSockets.
            juce::MessageManager::callAsync([this, ccNumber, ccValue, channel]() 
            {
                if (onCcReceivedFromHardware)
                    onCcReceivedFromHardware(ccNumber, ccValue, channel);
            });
        }
    }

    // --- WEBUI -> HARDWARE (Emisión) ---
    void sendCcToHardware(int ccNumber, int ccValue, int channel)
    {
        if (midiOutput != nullptr)
        {
            auto message = juce::MidiMessage::controllerEvent(channel, ccNumber, ccValue);
            midiOutput->sendMessageNow(message);
        }
    }

    // Métodos para conectar/desconectar los puertos físicos de Roland
    void setMidiDevices(std::unique_ptr<juce::MidiInput> input, std::unique_ptr<juce::MidiOutput> output)
    {
        midiInput = std::move(input);
        midiOutput = std::move(output);
        
        if (midiInput != nullptr)
            midiInput->start();
    }

    // Lambdas para comunicar con tu capa de WebSockets / WebUI
    std::function<void(int cc, int val, int ch)> onCcReceivedFromHardware;

private:
    std::unique_ptr<juce::MidiInput> midiInput;
    std::unique_ptr<juce::MidiOutput> midiOutput;
    double startTime;
};

class LabAudioReceiver
{
public:
    enum class ReceiverMode
    {
        Idle,
        CaptureTimeWindow, // Captura secuencial contando muestras (ADSR)
        CaptureSpectralBlock // Captura un bloque estático para FFT (Filtros)
    };

    LabAudioReceiver()
    {
        // Inicializar búfer de análisis con espacio de sobra (ej. 5 segundos a 44.1kHz)
        analysisBuffer.setSize(1, 44100 * 5);
        reset();
    }

    void reset()
    {
        currentMode = ReceiverMode::Idle;
        writeIndex = 0;
        isRecording = false;
        sampleCounter = 0;
        captureComplete = false;
    }

    void setMode(ReceiverMode newMode, int requiredSamples = 4096)
    {
        reset();
        currentMode = newMode;
        targetSamples = requiredSamples;
    }

    bool isCaptureComplete() const { return captureComplete; }
    
    // Acceso seguro al búfer de datos para el motor analítico
    const juce::AudioBuffer<float>& getCapturedData() const { return analysisBuffer; }
    int getMeasuredSamples() const { return sampleCounter; }

    // --- PROCESAMIENTO EN EL HILO DE AUDIO (INPUT) ---
    void processIncomingBlock(const juce::AudioBuffer<float>& inputBuffer)
    {
        if (currentMode == ReceiverMode::Idle || captureComplete)
            return;

        int numSamples = inputBuffer.getNumSamples();
        const float* inputData = inputBuffer.getReadPointer(0); // Entrada desde el hardware

        for (int i = 0; i < numSamples; ++i)
        {
            float currentSample = inputData[i];

            if (currentMode == ReceiverMode::CaptureTimeWindow)
            {
                // Disparo por umbral: Empezamos a grabar si supera el ruido de fondo
                if (!isRecording && std::abs(currentSample) > threshold)
                {
                    isRecording = true;
                    sampleCounter = 0;
                    writeIndex = 0;
                }

                if (isRecording)
                {
                    // Guardamos la muestra y avanzamos contadores
                    if (writeIndex < analysisBuffer.getNumSamples())
                    {
                        analysisBuffer.setSample(0, writeIndex++, currentSample);
                        sampleCounter++;
                    }

                    // Condición de parada (ej: si alcanza el límite de seguridad de muestras)
                    if (sampleCounter >= targetSamples)
                    {
                        isRecording = false;
                        captureComplete = true;
                        break;
                    }
                }
            }
            else if (currentMode == ReceiverMode::CaptureSpectralBlock)
            {
                // Grabación directa sin umbral (para filtros ya estabilizados)
                if (writeIndex < targetSamples)
                {
                    analysisBuffer.setSample(0, writeIndex++, currentSample);
                }
                
                if (writeIndex >= targetSamples)
                {
                    captureComplete = true;
                    break;
                }
            }
        }
    }

private:
    ReceiverMode currentMode = ReceiverMode::Idle;
    juce::AudioBuffer<float> analysisBuffer;
    
    int writeIndex = 0;
    int sampleCounter = 0;
    int targetSamples = 4096;
    
    bool isRecording = false;
    bool captureComplete = false;
    
    const float threshold = 0.01f; // Umbral de disparo aprox -40dB
};

#include <vector>
#include <numeric>

class LabAnalyticEngine
{
public:
    struct AnalysisResult
    {
        float stableValue = 0.0f; // La Media (Tiempo en ms o Frecuencia en Hz)
        float acbDrift = 0.0f;    // La Desviación Estándar (Intensidad del bamboleo)
    };

    LabAnalyticEngine() {}

    // --- ANÁLISIS PARA FILTROS (Mapeo espectral por ventanas) ---
    AnalysisResult analyzeFilterResponse(const juce::AudioBuffer<float>& capturedData, double sampleRate, int fftSize = 2048)
    {
        // Configuramos la FFT nativa de JUCE (requiere una potencia de 2, ej: 2048)
        int order = std::log2(fftSize);
        juce::dsp::FFT fft(order);
        
        // El ACB fluctúa en el tiempo. Vamos a dividir la captura larga en múltiples ventanas
        // cortas de FFT para ver cómo "tiembla" el filtro frame a frame.
        std::vector<float> peakFrequenciesInTime;
        int totalSamples = capturedData.getNumSamples();
        int hopSize = fftSize / 2; // 50% de solapamiento para mayor precisión
        
        std::vector<float> fftBuffer(fftSize * 2, 0.0f);
        const float* rawAudio = capturedData.getReadPointer(0);

        for (int offset = 0; offset + fftSize <= totalSamples; offset += hopSize)
        {
            // 1. Preparar la ventana de datos y aplicar un filtrado Hann para evitar artefactos
            std::fill(fftBuffer.begin(), fftBuffer.end(), 0.0f);
            for (int i = 0; i < fftSize; ++i)
                fftBuffer[i] = rawAudio[offset + i] * (0.5f * (1.0f - std::cos(2.0f * juce::MathConstants<float>::pi * i / (fftSize - 1))));

            // 2. Ejecutar la FFT de JUCE
            fft.performFrequencyOnlyForwardTransform(fftBuffer.data());

            // 3. Buscar cuál es el bin de frecuencia con más energía (el pico del filtro)
            int peakBin = 0;
            float maxEnergy = -1.0f;
            for (int bin = 0; bin < fftSize / 2; ++bin)
            {
                if (fftBuffer[bin] > maxEnergy)
                {
                    maxEnergy = fftBuffer[bin];
                    peakBin = bin;
                }
            }

            // Convertir el bin binario en frecuencia real en Hercios (Hz)
            float frequencyHz = (static_cast<float>(peakBin) * sampleRate) / static_cast<float>(fftSize);
            peakFrequenciesInTime.push_back(frequencyHz);
        }

        // 4. Calcular la Media y la Desviación Estándar de todas las ventanas analizadas
        return calculateStatisticalResult(peakFrequenciesInTime);
    }

    // --- ANÁLISIS PARA ENVOLVENTES (Tiempos de Ataque / Decay) ---
    AnalysisResult analyzeEnvelopeAttack(const juce::AudioBuffer<float>& capturedData, double sampleRate)
    {
        const float* rawAudio = capturedData.getReadPointer(0);
        int totalSamples = capturedData.getNumSamples();
        
        // Para capturar la inestabilidad del ataque del ACB, necesitaríamos lanzar múltiples impulsos.
        // Aquí analizamos el tiempo de ataque de la captura actual buscando el punto donde la amplitud se estabiliza.
        float currentPeak = 0.0f;
        int samplesToPeak = 0;
        float envelopeThreshold = 0.95f; // Buscamos cuando llegue al 95% del volumen máximo estable

        // Primero encontramos el pico absoluto real de la señal grabada
        for (int i = 0; i < totalSamples; ++i)
            if (std::abs(rawAudio[i]) > currentPeak) currentPeak = std::abs(rawAudio[i]);

        // Buscamos en qué muestra exacta cruza el umbral de estabilización
        for (int i = 0; i < totalSamples; ++i)
        {
            if (std::abs(rawAudio[i]) >= currentPeak * envelopeThreshold)
            {
                samplesToPeak = i;
                break;
            }
        }

        float timeInMs = (static_cast<float>(samplesToPeak) / sampleRate) * 1000.0f;
        
        AnalysisResult result;
        result.stableValue = timeInMs; // Tiempo medio medido en milisegundos
        result.acbDrift = 0.0f;        // Se rellenará promediando múltiples pasadas de ataques
        return result;
    }

private:
    // --- EL MOTOR ESTADÍSTICO ---
    AnalysisResult calculateStatisticalResult(const std::vector<float>& dataPoints)
    {
        AnalysisResult res;
        if (dataPoints.empty()) return res;

        // 1. Calcular la Media (μ)
        float sum = std::accumulate(dataPoints.begin(), dataPoints.end(), 0.0f);
        res.stableValue = sum / dataPoints.size();

        // 2. Calcular la Desviación Estándar (σ)
        float varianceSum = 0.0f;
        for (float val : dataPoints)
            varianceSum += std::pow(val - res.stableValue, 2);
        
        res.acbDrift = std::sqrt(varianceSum / dataPoints.size());

        return res;
    }
};

ESPECIFICACIONES DE LAS CAPAS C++ (EL CORE NATIVO)Capa 3.1: Gestión de Audio e Inyección de Estímulos (LabStimulusGenerator)Esta clase opera en el hilo de alta prioridad de la tarjeta de sonido dentro del entorno de drivers ASIO exclusivos bajo Windows.Responsabilidades:Soportar conmutación instantánea y síncrona entre estados de señal mediante un enumerado estricto: Silence, DiracDelta, LogFarinaSweep, WhiteNoise, PinkNoise, SineWave1kHz, SquareWave1kHz.Especificación del Farina Sweep: Generar una onda senoidal exponencial continua que barra desde exactamente 20 Hz hasta la frecuencia de Nyquist (SampleRate / 2) en una ventana temporal rígida dictada por el buffer de análisis.Principio Zero-Allocation: Prohibida la reserva o liberación de memoria dinámica dentro del método de procesamiento sample a sample. Toda la geometría de las ondas y punteros debe estar precalculada.Capa 3.2: Captura y Concurrencia Libre de Bloqueos (LabAudioReceiver)El receptor es la ventana que empaqueta el audio modificado que regresa del hardware analógico o digital.Responsabilidades:Implementar un Umbral de Disparo (Threshold Trigger) por amplitud nominal (ajustable por el JSON de perfil, por defecto -40 dBfs).Inmunidad a la Latencia: El contador de muestras de la prueba arranca únicamente cuando el valor absoluto de la muestra entrante supera el umbral, aislando por completo la latencia de ida y vuelta (Round-Trip Latency) introducida por el sistema operativo y el driver de la tarjeta de sonido.Estructura FIFO Lock-Free: El hilo de audio empuja las muestras capturadas hacia un búfer circular indexado por la clase juce::AbstractFifo. Ningún cálculo matemático ni FFT puede ejecutarse en este hilo para evitar cortes de audio (glitches).Capa 3.3: Cerebro Matemático y Extracción Estadística (LabAnalyticEngine)Esta clase se ejecuta en un hilo secundario de baja prioridad (Background Thread) que despierta en cuanto la estructura FIFO se llena con las muestras requeridas por el caso de prueba actual.Algoritmo de Extracción Temporal (Envolventes/Delays):Calcular la derivada de la envolvente de amplitud para localizar el punto exacto de estabilización de la señal.Convertir la distancia de muestras del contador en milisegundos reales: \(Tiempo_{ms} = (Samples / SampleRate) \times 1000\).Algoritmo de Extracción Espectral (Filtros/EQs):Segmentar la captura de audio en ventanas de tamaño potencia de 2 (ej. 2048 samples) aplicando un solapamiento (overlap) del 50%.Multiplicar el bloque por una Ventana de Hann para eliminar el emborronamiento espectral en los extremos antes de la transformada.Ejecutar la transformada rápida de Fourier mediante la clase nativa juce::dsp::FFT.Localizar el bin de máxima energía para determinar la frecuencia de pico en Hercios.Cálculo Dual de la Huella ACB/Analógica:Para cada ventana analizada dentro de un mismo estado estático, calcular la Media Matemática (μ) como el valor estático del control.Calcular la Desviación Estándar (σ) de todas las ventanas del mismo bloque como el factor de inestabilidad y bamboleo dinámico del hardware.4.
